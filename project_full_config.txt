    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        if (checkAndBlockSettingsAccess(event)) {
            return
        }

        val currentTime = System.currentTimeMillis()

        if (isOverlayShowing && (currentTime - lastNsfwDetectedTime < HOLD_MASK_MS)) {
            return
        }

        if (isCircuitBreakerActive || isAnalyzing) return

        val minInterval = if (isOverlayShowing) 500L else 350L
        if (currentTime - lastCaptureTime < minInterval) return

        val currentBounds = getVisibleImageBounds()
        val currentHash = currentBounds.hashCode()

        if (!isOverlayShowing && currentHash == lastImageNodesHash && currentBounds.isNotEmpty()) {
            return
        }

        lastImageNodesHash = currentHash
        lastCaptureTime = currentTime
        captureAndAnalyzeScreen(currentBounds)
    }

    private fun checkAndBlockSettingsAccess(event: AccessibilityEvent?): Boolean {
        val packageName = event?.packageName?.toString() ?: return false

        if (packageName == "com.android.settings" || packageName.contains("setting", ignoreCase = true)) {
            val rootNode = rootInActiveWindow ?: return false

            try {
                val hasAppName = isNodeTreeContainsText(rootNode, listOf("NSFWFilter", "nsfwfilter"))

                if (hasAppName) {
                    val isAppInfoPage = isNodeTreeContainsText(
                        rootNode,
                        listOf("DÃ©sinstaller", "Uninstall", "Forcer l'arrÃªt", "Force stop")
                    )

                    val isAccessibilityTogglePage = isNodeTreeContainsText(
                        rootNode,
                        listOf(
                            "Utiliser NSFWFilter", "Use NSFWFilter",
                            "Raccourci NSFWFilter", "NSFWFilter shortcut",
                            "Autoriser NSFWFilter", "Allow NSFWFilter"
                        )
                    ) || hasSwitchOrToggle(rootNode)

                    if (isAppInfoPage || isAccessibilityTogglePage) {
                        Log.w("FilterService", "--> TENTATIVE DE DÃ‰SACTIVATION DE NSFWFILTER : RETOUR ACCUEIL")
                        performGlobalAction(GLOBAL_ACTION_HOME)
                        return true
                    }
                }
            } finally {
                rootNode.recycle()
            }
        }
        return false
    }

    private fun hasSwitchOrToggle(node: AccessibilityNodeInfo?): Boolean {
        if (node == null) return false

        val className = node.className?.toString() ?: ""
        if (className.contains("Switch", ignoreCase = true) ||
            className.contains("ToggleButton", ignoreCase = true) ||
            className.contains("CheckBox", ignoreCase = true)) {
            return true
        }

        val text = node.text?.toString() ?: ""
        if (text.contains("Utiliser", ignoreCase = true) ||
            text.contains("Raccourci", ignoreCase = true) ||
            text.contains("Shortcut", ignoreCase = true)) {
            return true
        }

        for (i in 0 until node.childCount) {
            val child = node.getChild(i)
            if (hasSwitchOrToggle(child)) {
                child?.recycle()
                return true
            }
            child?.recycle()
        }
        return false
    }

    private fun isNodeTreeContainsText(node: AccessibilityNodeInfo?, keywords: List<String>): Boolean {
        if (node == null) return false

        val text = node.text?.toString() ?: ""
        val contentDesc = node.contentDescription?.toString() ?: ""

        for (keyword in keywords) {
            if (text.contains(keyword, ignoreCase = true) || contentDesc.contains(keyword, ignoreCase = true)) {
                return true
            }
        }

        for (i in 0 until node.childCount) {
            val child = node.getChild(i)
            if (isNodeTreeContainsText(child, keywords)) {
                child?.recycle()
                return true
            }
            child?.recycle()
        }
        return false
    }

    private fun captureAndAnalyzeScreen(imageBounds: List<Rect>) {
        isAnalyzing = true

        // SECURITÃ‰ ANTI-BLOCAGE : Si Android ne donne aucune rÃ©ponse au bout de 2 sec, on dÃ©bloque le flag
        watchdogJob?.cancel()
        watchdogJob = serviceScope.launch {
            delay(2000)
            if (isAnalyzing) {
                Log.w("FilterService", "Watchdog : capture sans rÃ©ponse du systÃ¨me, dÃ©blocage forcÃ©.")
                isAnalyzing = false
            }
        }

        val wasOverlayShowing = isOverlayShowing
        if (isOverlayShowing) {
            overlayView?.alpha = 0f
        }

        takeScreenshot(
            Display.DEFAULT_DISPLAY,
            screenshotExecutor,
            object : TakeScreenshotCallback {
                override fun onSuccess(screenshotResult: ScreenshotResult) {
                    watchdogJob?.cancel()
                    consecutiveFailures = 0
                    val hardwareBuffer = screenshotResult.hardwareBuffer
                    val colorSpace = screenshotResult.colorSpace

                    val fullBitmap = try {
                        Bitmap.wrapHardwareBuffer(hardwareBuffer, colorSpace)
                    } catch (e: Exception) {
                        null
                    }

                    if (wasOverlayShowing) {
                        serviceScope.launch(Dispatchers.Main) {
                            overlayView?.alpha = 1f
                        }
                    }

                    if (fullBitmap == null) {
                        hardwareBuffer.close()
                        isAnalyzing = false
                        return
                    }

                    val softwareBitmap = try {
                        fullBitmap.copy(Bitmap.Config.ARGB_8888, false)
                    } catch (e: Exception) {
                        null
                    }

                    fullBitmap.recycle()
                    hardwareBuffer.close()

                    if (softwareBitmap == null) {
                        isAnalyzing = false
                        return
                    }

                    serviceScope.launch {
                        try {
                            var isNsfwDetected = false

                            for (rect in imageBounds) {
                                val croppedBitmap = cropBitmap(softwareBitmap, rect) ?: continue
                                val cropScore = classifier.processAndClassify(croppedBitmap)
                                croppedBitmap.recycle()

                                if (cropScore > 0.45f) {
                                    isNsfwDetected = true
                                    Log.w("FilterService", "--> NSFW dÃ©tectÃ© zone $rect (Score: $cropScore)")
                                    break
                                }
                            }

                            if (!isNsfwDetected) {
                                val globalScore = classifier.processAndClassify(softwareBitmap)
                                if (globalScore > 0.55f) {
                                    isNsfwDetected = true
                                    Log.w("FilterService", "--> NSFW global dÃ©tectÃ© (Score: $globalScore)")
                                }
                            }

                            val currentTime = System.currentTimeMillis()
                            withContext(Dispatchers.Main) {
                                if (isNsfwDetected) {
                                    lastNsfwDetectedTime = currentTime
                                    val blurredBitmap = createSafeLightBlurBitmap(softwareBitmap)
                                    showOverlay(blurredBitmap)
                                } else {
                                    if (currentTime - lastNsfwDetectedTime > HOLD_MASK_MS) {
                                        hideOverlay()
                                    }
                                }
                            }
                        } catch (e: Exception) {
                            Log.e("FilterService", "Erreur analyse : ${e.message}", e)
                        } finally {
                            softwareBitmap.recycle()
                            isAnalyzing = false
                        }
                    }
                }

                override fun onFailure(errorCode: Int) {
                    watchdogJob?.cancel()
                    if (wasOverlayShowing) {
                        serviceScope.launch(Dispatchers.Main) {
                            overlayView?.alpha = 1f
                        }
                    }

                    consecutiveFailures++
                    Log.e("FilterService", "Ã‰chec capture (Code: $errorCode, ConsÃ©cutifs: $consecutiveFailures)")

                    if (consecutiveFailures >= 3) {
                        isCircuitBreakerActive = true
                        serviceScope.launch {
                            delay(4000)
                            consecutiveFailures = 0
                            isCircuitBreakerActive = false
                            isAnalyzing = false
                        }
                    } else {
                        serviceScope.launch {
                            delay(400)
                            isAnalyzing = false
                        }
                    }
                }
            }
        )

    private fun createSafeLightBlurBitmap(source: Bitmap): Bitmap {
        var tinyBitmap: Bitmap? = null
        var midBitmap: Bitmap? = null
        try {
            val lowWidth = (source.width * 0.05f).toInt().coerceAtLeast(1)
            val lowHeight = (source.height * 0.05f).toInt().coerceAtLeast(1)
            tinyBitmap = Bitmap.createScaledBitmap(source, lowWidth, lowHeight, true)

            val midWidth = (source.width * 0.15f).toInt().coerceAtLeast(1)
            val midHeight = (source.height * 0.15f).toInt().coerceAtLeast(1)
            midBitmap = Bitmap.createScaledBitmap(tinyBitmap, midWidth, midHeight, true)

            val blurredBitmap = Bitmap.createScaledBitmap(midBitmap, source.width, source.height, true)

            val canvas = Canvas(blurredBitmap)
            val lightPaint = Paint().apply {
                color = 0x55FFFFFF.toInt()
                style = Paint.Style.FILL
            }
            canvas.drawRect(0f, 0f, blurredBitmap.width.toFloat(), blurredBitmap.height.toFloat(), lightPaint)

            return blurredBitmap
        } finally {
            tinyBitmap?.recycle()
            midBitmap?.recycle()
        }
    }

    private fun getVisibleImageBounds(): List<Rect> {
        val root = rootInActiveWindow ?: return emptyList()
        val boundsList = mutableListOf<Rect>()
        try {
            findImageNodesRecursive(root, boundsList)
        } catch (e: Exception) {
            Log.e("FilterService", "Erreur lecture nÅ“uds : ${e.message}")
        } finally {
            root.recycle()
        }

        return boundsList
            .sortedByDescending { it.width() * it.height() }
            .take(6)
    }

    private fun findImageNodesRecursive(node: AccessibilityNodeInfo?, boundsList: MutableList<Rect>) {
        if (node == null) return

        val className = node.className?.toString() ?: ""
        val resourceId = node.viewIdResourceName?.lowercase() ?: ""

        val isImageOrMediaView = className.contains("Image", ignoreCase = true) ||
                className.contains("Graphic", ignoreCase = true) ||
                className.contains("View", ignoreCase = true) && (resourceId.contains("photo") || resourceId.contains("image") || resourceId.contains("media") || resourceId.contains("picture"))

        if (isImageOrMediaView) {
            val rect = Rect()
            node.getBoundsInScreen(rect)

            if (rect.width() > 80 && rect.height() > 80) {
                boundsList.add(rect)
            }
        }

        for (i in 0 until node.childCount) {
            val child = node.getChild(i)
            if (child != null) {
                findImageNodesRecursive(child, boundsList)
                child.recycle()
            }
        }
    }

    private fun cropBitmap(source: Bitmap, rect: Rect): Bitmap? {
        val x = rect.left.coerceIn(0, source.width - 1)
        val y = rect.top.coerceIn(0, source.height - 1)
        val width = rect.width().coerceAtMost(source.width - x)
        val height = rect.height().coerceAtMost(source.height - y)

        if (width <= 15 || height <= 15) return null
        return try {
            Bitmap.createBitmap(source, x, y, width, height)
        } catch (e: Exception) {
            null
        }
    }

    private fun showOverlay(blurredBitmap: Bitmap) {
        isOverlayShowing = true

        val oldDrawable = overlayView?.drawable as? BitmapDrawable
        overlayView?.setImageBitmap(blurredBitmap)
        oldDrawable?.bitmap?.recycle()

        overlayView?.alpha = 1f
    }

    private fun hideOverlay() {
        isOverlayShowing = false
        overlayView?.alpha = 0f

        val oldDrawable = overlayView?.drawable as? BitmapDrawable
        oldDrawable?.bitmap?.recycle()
        overlayView?.setImageBitmap(null)
    }

    override fun onInterrupt() {}

    override fun onDestroy() {
        super.onDestroy()
        watchdogJob?.cancel()
        serviceScope.cancel()
        screenshotExecutor.shutdown()
        if (isOverlayAttached && overlayView != null) {
            try {
                windowManager?.removeView(overlayView)
            } catch (e: Exception) {
                Log.e("FilterService", "Erreur retrait overlay : ${e.message}")
            }
        }
        classifier.close()
    }
}

package com.example.nsfwfilter

import android.content.Context
import android.graphics.Bitmap
import android.util.Log
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.support.common.ops.NormalizeOp
import org.tensorflow.lite.support.image.ImageProcessor
import org.tensorflow.lite.support.image.TensorImage
import org.tensorflow.lite.support.image.ops.ResizeOp
import java.io.FileInputStream
import java.nio.MappedByteBuffer
import java.nio.channels.FileChannel

class NSFWClassifier(private val context: Context) {

    private var interpreter: Interpreter? = null
    private var inferenceCount = 0

    init {
        initInterpreter()
    }

    private fun initInterpreter() {
        try {
            val options = Interpreter.Options().apply {
                setNumThreads(4)
            }
            val model = loadModelFile(context, "nsfw_model.tflite")
            interpreter = Interpreter(model, options)
        } catch (e: Exception) {
            Log.e("NSFWClassifier", "Erreur initialisation TFLite : ${e.message}")
        }
    }

    private fun loadModelFile(context: Context, fileName: String): MappedByteBuffer {
        val fileDescriptor = context.assets.openFd(fileName)
        val inputStream = FileInputStream(fileDescriptor.fileDescriptor)
        val fileChannel = inputStream.channel
        return fileChannel.map(
            FileChannel.MapMode.READ_ONLY,
            fileDescriptor.startOffset,
            fileDescriptor.declaredLength
        )
    }

    @Synchronized
    fun processAndClassify(bitmap: Bitmap): Float {
        inferenceCount++
        if (inferenceCount >= 300) {
            recycleNativeMemory()
            inferenceCount = 0
        }

        val currentInterpreter = interpreter ?: return 0.0f

        if (!hasSkinTone(bitmap)) {
            return 0.0f
        }

        return try {
            val imageProcessor = ImageProcessor.Builder()
                .add(ResizeOp(224, 224, ResizeOp.ResizeMethod.BILINEAR))
